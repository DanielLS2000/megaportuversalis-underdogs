# Sindicat Remença
country_event = {
	id = flavor_afr.14
	title = flavor_afr.14.t
	desc = flavor_afr.14.d
	picture = ELECTION_REPUBLICAN_eventPicture
	
	fire_only_once = yes
	
	trigger = {
		tag = AFR
		catalonia_area = {
			type = all 
			country_or_non_sovereign_subject_holds = ROOT 
		}
	}
	
	mean_time_to_happen = {
		months = 60
	}
	
	immediate = {
		hidden_effect = {
			set_country_flag = afr_remensa_flag
			random_owned_province = {
				limit = {
					is_core = ROOT
					culture = catalan
				}
				set_province_flag = afr_sindicat_remensa_flag
			}
			random_owned_province = {
				limit = {
					is_core = ROOT
					culture = catalan
					NOT = {  has_province_flag = afr_sindicat_remensa_flag }
				}
				set_province_flag = afr_sindicat_remensa_flag
			}
			random_owned_province = {
				limit = {
					is_core = ROOT
					culture = catalan
					NOT = { has_province_flag = afr_sindicat_remensa_flag }
				}
				set_province_flag = afr_sindicat_remensa_flag
			}
		}
	}
	
	option = {
		name = flavor_afr.14.a
		trigger = {
			if = { 
				limit = { has_dlc = "Domination" }
				has_estate = estate_burghers
			}
		}
		add_country_modifier = {
			name = "decentralization_modifier"
			duration = 3650
		}
		if = {
			limit = { has_dlc = "Domination" }
			set_country_flag = remensa_privilege_flag
			set_estate_privilege = estate_burghers_sindicat_remensa_privilege
		}
		else = {
			every_owned_province = {
				limit = {
					has_province_flag = afr_sindicat_remensa_flag
				}
				add_province_modifier = { 
					name = afr_sindicat_remensa 
					duration = -1
				}
			}
			hidden_effect = {
				every_owned_province = {
					limit = {
						has_province_flag = afr_sindicat_remensa_flag
					}
					clr_province_flag = afr_sindicat_remensa_flag
				}
			}
		}
		ai_chance = {
			factor = 45
		}
	}
	option = {
		name = flavor_afr.14.b
		every_owned_province = {
			limit = {
				has_province_flag = afr_sindicat_remensa_flag
			}
			clr_province_flag = afr_sindicat_remensa_flag
			spawn_rebels = {
				type = anti_tax_rebels
				size = 2
			}
			add_local_autonomy = -25
		}
		ai_chance = {
			factor = 45
		}
	}
	
	option = {
		name = flavor_afr.14.c
		
		if = { 
			limit = { 
				stability = 0 
			}
			add_stability = -3 
		}
		else = { 
			add_adm_power = -50 
			add_dip_power = -50
			add_mil_power = -50
		}
		every_owned_province = {
			limit = {
				has_province_flag = afr_sindicat_remensa_flag
			}
			clr_province_flag = afr_sindicat_remensa_flag
			spawn_rebels = {
				type = noble_rebels
				size = 3
			}
		}
		define_ruler = {
			name = "Èric" 
			DIP = 6
			ADM = 5
			MIL = 4
			age = 28
			
		}
		if = {
			limit = { has_dlc = "Rights of Man" }
			add_ruler_personality = careful_personality
			add_ruler_personality = peasant_revolt_leader_personality
		}
		custom_tooltip = afr_become_p_republic_tt
		add_stability_or_adm_power = yes 
		hidden_effect = {
			change_government = republic 
			add_government_reform = peasants_republic
			hidden_effect = { add_republican_tradition = 100 }
		}
		ai_chance = {
			factor = 10
		}
	}
}
