# Renaissance Centre
country_event = {
	id = flavor_afr.1
	title = flavor_afr.1.t
	desc = flavor_afr.1.d
	picture = IMPORTANT_STATUE_eventPicture
	
	fire_only_once = yes

	trigger = {
		OR = {
			has_country_flag = naples_renaissance_flag
			AND = {
				NOT = { has_mission = emp_nap_renaissance_city }
				OR = {
					tag = AFR
					tag = ITA
				}
				is_year = 1460
				NOT = { is_year = 1530 }
				stability = 3
				is_at_war = no
			}
		}
	}
	
	mean_time_to_happen = {
		months = 4
	}	
	
	option = {
		name = flavor_afr.1.a
		add_prestige = 10
		if = {
			limit = {
				NOT = { has_completed_all_reforms_trigger = yes }
			}
			add_reform_progress_small_effect = yes
		}
		add_country_modifier = {
			name = "afr_renaissance_culture"
			duration = 18250
		}
	}
	option = {
		name = flavor_afr.1.b
		add_legitimacy = 10
		if = {
			limit = {
				NOT = { has_completed_all_reforms_trigger = yes }
			}
			add_reform_progress_small_effect = yes
		}
		add_country_modifier = {
			name = "afr_renaissance_glory"
			duration = 18250
		}
	}
}

# Consulate of the Sea
country_event = {
	id = flavor_afr.3
	title = flavor_afr.3.t
	desc = flavor_afr.3.d
	picture = COURT_eventPicture
	
	fire_only_once = yes
	
	trigger = {
		OR = {
			AND = {
				NOT = { has_dlc = "Golden Century" }
				tag = AFR
				NOT = { is_year = 1500 }
				NOT = {
					has_country_modifier = afr_consulate_sea
				}
			}
			has_country_flag = afr_consulate_flag
		}
	}
	
	mean_time_to_happen = {
		months = 200
	}
	
	option = {
		name = flavor_afr.3.a
		if = {
			limit = { has_dlc = "Domination" }
			add_government_reform = consulate_of_the_sea
		}
		else = {
			add_country_modifier = {
				name = afr_consulate_sea
				duration = 18250
			}
		}
	}
}
