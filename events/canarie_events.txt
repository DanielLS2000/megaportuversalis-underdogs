# Consolato del Mar
country_event = {
	id = flavor_cnr.1
	title = "flavor_cnr.title"
	desc = "flavor_cnr.desc"
	picture = COLONIZATION_eventPicture
	
	fire_only_once = yes
	
	trigger = {
		OR = {
			AND = {
				NOT = { has_dlc = "Golden Century" }
				OR = {
					AND = {
						tag = CNR
						NOT = { exists = MEX }
					}
					tag = CNR
				}
				num_of_colonists = 2
			}
			has_country_flag = cnr_house_of_trade_flag
		}
	}

	mean_time_to_happen = {
		months = 120
	}


	option = {		# We should establish it
		name = flavor_cnr.1
		ai_chance = { factor = 90 }
		add_country_modifier = {
			name = consolato_del_mar
			duration = -1
		}
		clr_country_flag = cnr_house_of_trade_flag
	}
}

			if = {
				limit = { has_building = shipyard }
				remove_building = shipyard
				add_building_construction = {
					building = grand_shipyard
				}
			}
			else_if = {
				limit = { has_building = grand_shipyard }

		}

			duration = -1
		}
	}
}
