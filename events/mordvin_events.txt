# The yearly Tatar Yoke Tribute
country_event = {
	id = flavor_mdv.4
	title = flavor_mdv.4.t
	desc = flavor_mdv.4.desc
	picture = TRIBUTE_TO_TATARS_eventPicture
	is_triggered_only = yes

	trigger = {
		has_dlc = "Domination"
		NOT = { has_global_flag = end_of_the_tatar_yoke }
		has_country_modifier = the_tatar_yoke
		466 = {
			owner = {
				NOT = { war_with = ROOT }
			}
		}
	}

	mean_time_to_happen = {
		days = 1
	}
    
    immediate = {
        hidden_effect = {
            export_to_variable = {
                which = mos_tribute_to_pay
                value = monthly_income
                who = ROOT
            }
            multiply_variable = {
                which = mos_tribute_to_pay
                value = 2
            }
        }
    }

	#Pay the Annual Tribute
	option = {
		name = flavor_mdv.4.a
		custom_tooltip = flavor_mdv.4.a.tt
		hidden_effect = {
			466 = {
				owner = {
					country_event = {
						id = flavor_mdv.5
					}
				}
			}
		}
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0
				OR = {
					NOT = { years_of_income = 0.3 }
					466 = {
						owner = {
							OR = {
								is_at_war = yes
								NOT = { manpower_percentage = 0.5 }
								NOT = { mil_power_cost = 50 }
							}
						}
					}
				}
			}
		}
	}

	#Refuse to pay the Annual Tribute
	option = {
		name = flavor_mdv.4.b
		custom_tooltip = flavor_mdv.4.b.tt
		hidden_effect = {
			466 = {
				owner = {
					country_event = {
						id = flavor_mdv.6
					}
				}
			}
		}
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0
				years_of_income = 0.3
				466 = {
					owner = {
						is_at_war = no
						manpower_percentage = 0.5
						mil_power_cost = 50
					}
				}
			}
		}
	}
}

	desc = flavor_mdv.8.desc

		has_dlc = "Domination"

		has_ruler_flag = has_vasily_ii
		num_of_cities = 2
	}
	
	mean_time_to_happen = {
		months = 50
	}
	
	option = {
		name = flavor_mdv.8.a
		exile_ruler_as = {
			name = vasily_ii
		}
		exile_heir_as = ivan_veliky
		hidden_effect = {
			if = {
				limit = { has_leader = "Puresh" }
				define_leader_to_ruler = {
					name = "Puresh"
					adm = 2
					dip = 1
					mil = 5
					claim = 40
				}
				kill_leader = {
					type = "Puresh"
				}
			}
			else = {
				define_ruler = {
					name = "Puresh"
					dynasty = "Erzya"
					min_age = 20
					max_age = 40
					adm = 2
					dip = 1
					mil = 5
					claim = 40
				}
			}
			set_ruler_flag = dmitry_shemiaka_in_power
		}
		add_ruler_modifier = {
			name = vasily_ii_in_exile
			duration = -1
			desc = vasily_ii_in_exile_desc
		}
		
		ai_chance = {
			factor = 1
		}
	}
}
